# Fixed calculate_fairness_metrics_full function
# ===================================================
def calculate_fairness_metrics_full(X_data, y_data, y_pred, sensitive_feature):
    groups = X_data[sensitive_feature].unique()
    tprs = []
    fprs = []
    parity_rates = []

    for group in groups:
        group_idx = X_data[sensitive_feature] == group
        group_y_data = y_data[group_idx]
        group_y_pred = y_pred[group_idx]

        if len(group_y_data) == 0:
            continue

        # Calculate TPR (recall)
        group_tpr = recall_score(group_y_data, group_y_pred, zero_division=0)
        tprs.append(group_tpr)

        # Calculate FPR with proper handling
        try:
            # Use labels to ensure 2x2 matrix even with single class
            cm = confusion_matrix(group_y_data, group_y_pred, labels=[0, 1])
            tn, fp, fn, tp = cm.ravel()
            group_fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
        except ValueError:
            # Fallback in case of issues
            group_fpr = 0
        fprs.append(group_fpr)

        # Calculate positive prediction rate
        group_parity = group_y_pred.mean()
        parity_rates.append(group_parity)

    # Handle cases with no valid groups
    if not tprs:
        tprs = [0]
    if not parity_rates:
        parity_rates = [0]
    if not fprs:
        fprs = [0]

    # FIX: Ensure we have at least 2 groups for differences
    eod = max(tprs) - min(tprs) if len(tprs) > 1 else 0
    spd = max(parity_rates) - min(parity_rates) if len(parity_rates) > 1 else 0
    eodd = (max(tprs) - min(tprs)) + (max(fprs) - min(fprs)) if (len(tprs) > 1 and len(fprs) > 1) else 0
    fairness_ratio = eod / spd if spd != 0 else 0

    return {
        'TPR': tprs,
        'FPR': fprs,
        'Parity_Rate': parity_rates,
        'Equal_Opportunity_Diff': eod,
        'Statistical_Parity_Diff': spd,
        'Equalised_Odds_Difference': eodd,
        'Fairness_Ratio': fairness_ratio
    }

# SHAP Consistency Calculation Function (FIXED)
# ===================================================
def calculate_shap_consistency_gap(model, X_sample, sensitive_features):
    """Calculate max SHAP consistency gap across sensitive features"""
    explainer = shap.TreeExplainer(model)
    shap_values_all = explainer.shap_values(X_sample)

    # Handle SHAP values extraction
    if isinstance(shap_values_all, list):
        # For binary classification, use class 1 SHAP values
        shap_values = shap_values_all[1]
    elif len(shap_values_all.shape) == 3:
        # Handle 3D array (samples, features, classes)
        shap_values = shap_values_all[:, :, 1]  # Use class 1 values
    else:
        shap_values = shap_values_all

    # Ensure SHAP values are 2D array
    if len(shap_values.shape) == 1:
        shap_values = shap_values.reshape(1, -1)
    elif len(shap_values.shape) == 3:
        # Flatten last two dimensions if needed
        shap_values = shap_values.reshape(shap_values.shape[0], -1)

    n_samples_shap = shap_values.shape[0]
    max_gap = 0

    for feat in sensitive_features:
        groups = X_sample[feat].unique()
        group_consistencies = []

        for group in groups:
            # Create boolean mask for group
            group_mask = (X_sample[feat] == group).values

            # Trim mask to match SHAP values length if needed
            if len(group_mask) > n_samples_shap:
                group_mask = group_mask[:n_samples_shap]

            # Only proceed if we have samples to process
            if group_mask.sum() == 0:
                continue

            group_shap = shap_values[group_mask]

            # Skip if not enough samples for similarity calculation
            if len(group_shap) < 2:
                continue

            # Ensure array is 2D before cosine similarity
            if len(group_shap.shape) > 2:
                group_shap = group_shap.reshape(group_shap.shape[0], -1)

            # Compute pairwise cosine similarities
            sim_matrix = cosine_similarity(group_shap)
            np.fill_diagonal(sim_matrix, np.nan)
            group_consistency = np.nanmean(sim_matrix)
            group_consistencies.append(group_consistency)

        if len(group_consistencies) < 2:
            continue

        gap = max(group_consistencies) - min(group_consistencies)
        if gap > max_gap:
            max_gap = gap

    return max_gap

def objective(trial):
    """Objective function for multi-objective Optuna optimization"""
    params = {
        'n_estimators': trial.suggest_int('n_estimators', 100, 500),
        'max_depth': trial.suggest_int('max_depth', 5, 30),
        'min_samples_split': trial.suggest_int('min_samples_split', 2, 20),
        'min_samples_leaf': trial.suggest_int('min_samples_leaf', 1, 20),
        'max_features': trial.suggest_categorical('max_features', ['sqrt', 'log2', None]),
        'bootstrap': trial.suggest_categorical('bootstrap', [True, False]),
        'class_weight': trial.suggest_categorical('class_weight', [None, 'balanced']),
        'random_state': 42,
        'n_jobs': -1
    }

    # Use 3-fold CV for optimization
    n_folds = 3
    skf = StratifiedKFold(n_splits=n_folds, shuffle=True, random_state=42)

    mcc_scores = []
    eodd_scores = []
    shap_gap_scores = []

    for fold, (train_idx, val_idx) in enumerate(skf.split(X_train_res, y_train_res)):
        X_fold_train, X_fold_val = X_train_res.iloc[train_idx], X_train_res.iloc[val_idx]
        y_fold_train, y_fold_val = y_train_res.iloc[train_idx], y_train_res.iloc[val_idx]

        model = RandomForestClassifier(**params)
        model.fit(X_fold_train, y_fold_train)

        # Performance: MCC
        y_pred_val = model.predict(X_fold_val)
        mcc = matthews_corrcoef(y_fold_val, y_pred_val)
        mcc_scores.append(mcc)

        # Fairness: 1 - EODD (maximize fairness = minimize EODD)
        avg_eodd = 0
        valid_features = 0
        for feat in sensitive_features:
            try:
                fairness = calculate_fairness_metrics_full(X_fold_val, y_fold_val, y_pred_val, feat)
                # FIX: Use Equalised_Odds_Difference instead of Equal_Opportunity_Diff
                avg_eodd += fairness['Equalised_Odds_Difference']
                valid_features += 1
            except Exception as e:
                # Skip feature if calculation fails
                print(f"Fairness calculation error for {feat}: {str(e)}")
                continue

        if valid_features > 0:
            avg_eodd /= valid_features
        else:
            avg_eodd = 0  # Default if no valid features
        eodd_scores.append(avg_eodd)

        # Explainability: 1 - SHAP_Consistency_Gap
        # Sample 100 instances for SHAP computation
        if len(X_fold_val) > 100:
            sample_indices = np.random.choice(X_fold_val.index, 100, replace=False)
            X_sample = X_fold_val.loc[sample_indices]
        else:
            X_sample = X_fold_val

        shap_gap = calculate_shap_consistency_gap(model, X_sample, sensitive_features)
        shap_gap_scores.append(shap_gap)

    # Aggregate scores across folds
    avg_mcc = np.mean(mcc_scores)
    avg_eodd = np.mean(eodd_scores)
    avg_shap_gap = np.mean(shap_gap_scores)

    # Return objectives: maximize MCC, maximize (1 - EODD), maximize (1 - SHAP_Gap)
    return avg_mcc, 1 - avg_eodd, 1 - avg_shap_gap

# Calculate Confidence Intervals (95%)
# ===================================================
def calculate_ci(data, confidence=0.95):
    a = 1.0 * np.array(data)
    n = len(a)
    m, se = np.mean(a), stats.sem(a)
    h = se * stats.t.ppf((1 + confidence) / 2., n-1)
    return m, m-h, m+h

    perf_ci[metric] = {'mean': mean, 'lower': lower, 'upper': upper}

# Fixed calculate_fairness_metrics_full function
# ===================================================
def calculate_fairness_metrics_full(X_data, y_data, y_pred, sensitive_feature):
    groups = X_data[sensitive_feature].unique()
    tprs = []
    fprs = []
    parity_rates = []

    for group in groups:
        group_idx = X_data[sensitive_feature] == group
        group_y_data = y_data[group_idx]
        group_y_pred = y_pred[group_idx]

        if len(group_y_data) == 0:
            continue

        # Calculate TPR (recall)
        group_tpr = recall_score(group_y_data, group_y_pred, zero_division=0)
        tprs.append(group_tpr)

        # Calculate FPR with proper handling
        try:
            # Use labels to ensure 2x2 matrix even with single class
            cm = confusion_matrix(group_y_data, group_y_pred, labels=[0, 1])
            tn, fp, fn, tp = cm.ravel()
            group_fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
        except ValueError:
            # Fallback in case of issues
            group_fpr = 0
        fprs.append(group_fpr)

        # Calculate positive prediction rate
        group_parity = group_y_pred.mean()
        parity_rates.append(group_parity)

    # Handle cases with no valid groups
    if not tprs:
        tprs = [0]
    if not parity_rates:
        parity_rates = [0]
    if not fprs:
        fprs = [0]

    # FIX: Ensure we have at least 2 groups for differences
    eod = max(tprs) - min(tprs) if len(tprs) > 1 else 0
    spd = max(parity_rates) - min(parity_rates) if len(parity_rates) > 1 else 0
    eodd = (max(tprs) - min(tprs)) + (max(fprs) - min(fprs)) if (len(tprs) > 1 and len(fprs) > 1) else 0
    fairness_ratio = eod / spd if spd != 0 else 0

    return {
        'TPR': tprs,
        'FPR': fprs,
        'Parity_Rate': parity_rates,
        'Equal_Opportunity_Diff': eod,
        'Statistical_Parity_Diff': spd,
        'Equalised_Odds_Difference': eodd,
        'Fairness_Ratio': fairness_ratio
    }

# SHAP Consistency Calculation Function (FIXED)
# ===================================================
def calculate_shap_consistency_gap(model, X_sample, sensitive_features):
    """Calculate max SHAP consistency gap across sensitive features"""
    explainer = shap.TreeExplainer(model)
    shap_values_all = explainer.shap_values(X_sample)

    # Handle SHAP values extraction
    if isinstance(shap_values_all, list):
        # For binary classification, use class 1 SHAP values
        shap_values = shap_values_all[1]
    elif len(shap_values_all.shape) == 3:
        # Handle 3D array (samples, features, classes)
        shap_values = shap_values_all[:, :, 1]  # Use class 1 values
    else:
        shap_values = shap_values_all

    # Ensure SHAP values are 2D array
    if len(shap_values.shape) == 1:
        shap_values = shap_values.reshape(1, -1)
    elif len(shap_values.shape) == 3:
        # Flatten last two dimensions if needed
        shap_values = shap_values.reshape(shap_values.shape[0], -1)

    n_samples_shap = shap_values.shape[0]
    max_gap = 0

    for feat in sensitive_features:
        groups = X_sample[feat].unique()
        group_consistencies = []

        for group in groups:
            # Create boolean mask for group
            group_mask = (X_sample[feat] == group).values

            # Trim mask to match SHAP values length if needed
            if len(group_mask) > n_samples_shap:
                group_mask = group_mask[:n_samples_shap]

            # Only proceed if we have samples to process
            if group_mask.sum() == 0:
                continue

            group_shap = shap_values[group_mask]

            # Skip if not enough samples for similarity calculation
            if len(group_shap) < 2:
                continue

            # Ensure array is 2D before cosine similarity
            if len(group_shap.shape) > 2:
                group_shap = group_shap.reshape(group_shap.shape[0], -1)

            # Compute pairwise cosine similarities
            sim_matrix = cosine_similarity(group_shap)
            np.fill_diagonal(sim_matrix, np.nan)
            group_consistency = np.nanmean(sim_matrix)
            group_consistencies.append(group_consistency)

        if len(group_consistencies) < 2:
            continue

        gap = max(group_consistencies) - min(group_consistencies)
        if gap > max_gap:
            max_gap = gap

    return max_gap

def objective(trial):
    """Objective function for multi-objective Optuna optimization"""
    # CHANGED: XGBoost hyperparameters instead of Random Forest
    params = {
        'n_estimators': trial.suggest_int('n_estimators', 100, 500),
        'max_depth': trial.suggest_int('max_depth', 3, 15),
        'learning_rate': trial.suggest_float('learning_rate', 0.01, 0.3, log=True),
        'min_child_weight': trial.suggest_int('min_child_weight', 1, 10),
        'subsample': trial.suggest_float('subsample', 0.5, 1.0),
        'colsample_bytree': trial.suggest_float('colsample_bytree', 0.5, 1.0),
        'gamma': trial.suggest_float('gamma', 0, 1),
        'reg_alpha': trial.suggest_float('reg_alpha', 0, 1),
        'reg_lambda': trial.suggest_float('reg_lambda', 0, 1),
        'random_state': 42,
        'n_jobs': -1,
        'use_label_encoder': False,
        'eval_metric': 'logloss'
    }

    # Use 3-fold CV for optimization
    n_folds = 3
    skf = StratifiedKFold(n_splits=n_folds, shuffle=True, random_state=42)

    mcc_scores = []
    eodd_scores = []
    shap_gap_scores = []

    for fold, (train_idx, val_idx) in enumerate(skf.split(X_train_res, y_train_res)):
        X_fold_train, X_fold_val = X_train_res.iloc[train_idx], X_train_res.iloc[val_idx]
        y_fold_train, y_fold_val = y_train_res.iloc[train_idx], y_train_res.iloc[val_idx]

        # CHANGED: Use XGBClassifier instead of RandomForestClassifier
        model = XGBClassifier(**params)
        model.fit(X_fold_train, y_fold_train)

        # Performance: MCC
        y_pred_val = model.predict(X_fold_val)
        mcc = matthews_corrcoef(y_fold_val, y_pred_val)
        mcc_scores.append(mcc)

        # Fairness: 1 - EODD (maximize fairness = minimize EODD)
        avg_eodd = 0
        valid_features = 0
        for feat in sensitive_features:
            try:
                fairness = calculate_fairness_metrics_full(X_fold_val, y_fold_val, y_pred_val, feat)
                # FIX: Use Equalised_Odds_Difference instead of Equal_Opportunity_Diff
                avg_eodd += fairness['Equalised_Odds_Difference']
                valid_features += 1
            except Exception as e:
                # Skip feature if calculation fails
                print(f"Fairness calculation error for {feat}: {str(e)}")
                continue

        if valid_features > 0:
            avg_eodd /= valid_features
        else:
            avg_eodd = 0  # Default if no valid features
        eodd_scores.append(avg_eodd)

        # Explainability: 1 - SHAP_Consistency_Gap
        # Sample 100 instances for SHAP computation
        if len(X_fold_val) > 100:
            sample_indices = np.random.choice(X_fold_val.index, 100, replace=False)
            X_sample = X_fold_val.loc[sample_indices]
        else:
            X_sample = X_fold_val

        shap_gap = calculate_shap_consistency_gap(model, X_sample, sensitive_features)
        shap_gap_scores.append(shap_gap)

    # Aggregate scores across folds
    avg_mcc = np.mean(mcc_scores)
    avg_eodd = np.mean(eodd_scores)
    avg_shap_gap = np.mean(shap_gap_scores)

    # Return objectives: maximize MCC, maximize (1 - EODD), maximize (1 - SHAP_Gap)
    return avg_mcc, 1 - avg_eodd, 1 - avg_shap_gap

# Calculate Confidence Intervals (95%)
# ===================================================
def calculate_ci(data, confidence=0.95):
    a = 1.0 * np.array(data)
    n = len(a)
    m, se = np.mean(a), stats.sem(a)
    h = se * stats.t.ppf((1 + confidence) / 2., n-1)
    return m, m-h, m+h

        sorted_vals = np.sort(np.array(list(values)))
        n = len(sorted_vals)
        index = np.arange(1, n+1)
        return (np.sum((2 * index - n - 1) * sorted_vals)) / (n * np.sum(sorted_vals))

    pre_gini = gini_coeff(list(pre_smote_distributions[feat].values()))
    post_gini = gini_coeff(list(post_smote_distributions[feat].values()))
    gini_change = post_gini - pre_gini

# Fixed calculate_fairness_metrics_full function
# ===================================================
def calculate_fairness_metrics_full(X_data, y_data, y_pred, sensitive_feature):
    groups = X_data[sensitive_feature].unique()
    tprs = []
    fprs = []
    parity_rates = []

    for group in groups:
        group_idx = X_data[sensitive_feature] == group
        group_y_data = y_data[group_idx]
        group_y_pred = y_pred[group_idx]

        if len(group_y_data) == 0:
            continue

        # Calculate TPR (recall)
        group_tpr = recall_score(group_y_data, group_y_pred, zero_division=0)
        tprs.append(group_tpr)

        # Calculate FPR with proper handling
        try:
            # Use labels to ensure 2x2 matrix even with single class
            cm = confusion_matrix(group_y_data, group_y_pred, labels=[0, 1])
            tn, fp, fn, tp = cm.ravel()
            group_fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
        except ValueError:
            # Fallback in case of issues
            group_fpr = 0
        fprs.append(group_fpr)

        # Calculate positive prediction rate
        group_parity = group_y_pred.mean()
        parity_rates.append(group_parity)

    # Handle cases with no valid groups
    if not tprs:
        tprs = [0]
    if not parity_rates:
        parity_rates = [0]
    if not fprs:
        fprs = [0]

    # FIX: Ensure we have at least 2 groups for differences
    eod = max(tprs) - min(tprs) if len(tprs) > 1 else 0
    spd = max(parity_rates) - min(parity_rates) if len(parity_rates) > 1 else 0
    eodd = (max(tprs) - min(tprs)) + (max(fprs) - min(fprs)) if (len(tprs) > 1 and len(fprs) > 1) else 0
    fairness_ratio = eod / spd if spd != 0 else 0

    return {
        'TPR': tprs,
        'FPR': fprs,
        'Parity_Rate': parity_rates,
        'Equal_Opportunity_Diff': eod,
        'Statistical_Parity_Diff': spd,
        'Equalised_Odds_Difference': eodd,
        'Fairness_Ratio': fairness_ratio
    }

# SHAP Consistency Calculation Function for TabNet
# ===================================================
def calculate_shap_consistency_gap_tabnet(model, X_sample, sensitive_features):
    """Calculate max SHAP consistency gap across sensitive features for TabNet"""
    # Use KernelExplainer for TabNet
    explainer = shap.KernelExplainer(model.predict_proba, X_sample.sample(50, random_state=42))
    shap_values = explainer.shap_values(X_sample, silent=True)

    # Use SHAP values for positive class
    if isinstance(shap_values, list):
        shap_values = shap_values[1]  # For binary classification
    else:
        shap_values = shap_values[:, :, 1] if len(shap_values.shape) == 3 else shap_values

    n_samples_shap = shap_values.shape[0]
    max_gap = 0

    for feat in sensitive_features:
        groups = X_sample[feat].unique()
        group_consistencies = []

        for group in groups:
            # Create boolean mask for group
            group_mask = (X_sample[feat] == group).values

            # Trim mask to match SHAP values length if needed
            if len(group_mask) > n_samples_shap:
                group_mask = group_mask[:n_samples_shap]

            # Only proceed if we have samples to process
            if group_mask.sum() == 0:
                continue

            group_shap = shap_values[group_mask]

            # Skip if not enough samples for similarity calculation
            if len(group_shap) < 2:
                continue

            # Ensure array is 2D before cosine similarity
            if len(group_shap.shape) > 2:
                group_shap = group_shap.reshape(group_shap.shape[0], -1)

            # Compute pairwise cosine similarities
            sim_matrix = cosine_similarity(group_shap)
            np.fill_diagonal(sim_matrix, np.nan)
            group_consistency = np.nanmean(sim_matrix)
            group_consistencies.append(group_consistency)

        if len(group_consistencies) < 2:
            continue

        gap = max(group_consistencies) - min(group_consistencies)
        if gap > max_gap:
            max_gap = gap

    return max_gap

def objective(trial):

        'lr': trial.suggest_float('lr', 1e-3, 0.1, log=True),
        'batch_size': trial.suggest_categorical('batch_size', [256, 512, 1024]),
        'virtual_batch_size': trial.suggest_categorical('virtual_batch_size', [64, 128, 256])
    }

    # Use 3-fold CV for optimization
    n_folds = 3
    skf = StratifiedKFold(n_splits=n_folds, shuffle=True, random_state=42)

    mcc_scores = []
    eodd_scores = []
    shap_gap_scores = []

    for fold, (train_idx, val_idx) in enumerate(skf.split(X_train_res, y_train_res)):
        print(f"  > Fold {fold+1}/{n_folds}")
        X_fold_train, X_fold_val = X_train_res.iloc[train_idx], X_train_res.iloc[val_idx]
        y_fold_train, y_fold_val = y_train_res.iloc[train_idx], y_train_res.iloc[val_idx]

        # Convert to numpy for TabNet
        X_fold_train_np = X_fold_train.values
        y_fold_train_np = y_fold_train.values
        X_fold_val_np = X_fold_val.values

        # Create TabNet model with suggested parameters
        model = TabNetClassifier(
            n_d=params['n_d'],
            n_a=params['n_a'],
            n_steps=params['n_steps'],
            gamma=params['gamma'],
            lambda_sparse=params['lambda_sparse'],
            optimizer_fn=torch.optim.Adam,
            optimizer_params=dict(lr=params['lr']),
            scheduler_params=dict(mode="min", patience=5, min_lr=1e-5, factor=0.9),
            scheduler_fn=torch.optim.lr_scheduler.ReduceLROnPlateau,
            mask_type='sparsemax',
            device_name='cuda' if torch.cuda.is_available() else 'cpu',
            seed=42,
            verbose=0
        )

        # Train model
        model.fit(
            X_fold_train_np, y_fold_train_np,
            eval_set=[(X_fold_val_np, y_fold_val.values)],
            eval_name=['val'],
            eval_metric=['auc'],
            max_epochs=100,
            patience=10,
            batch_size=params['batch_size'],
            virtual_batch_size=params['virtual_batch_size'],
            num_workers=0,
            drop_last=False,
            loss_fn=torch.nn.functional.cross_entropy
        )

        # Performance: MCC
        y_pred_val = model.predict(X_fold_val_np)
        mcc = matthews_corrcoef(y_fold_val, y_pred_val)
        mcc_scores.append(mcc)

        # Fairness: 1 - EODD (maximize fairness = minimize EODD)
        avg_eodd = 0
        valid_features = 0
        for feat in sensitive_features:
            try:
                fairness = calculate_fairness_metrics_full(X_fold_val, y_fold_val, y_pred_val, feat)
                avg_eodd += fairness['Equalised_Odds_Difference']
                valid_features += 1
            except Exception as e:
                print(f"Fairness calculation error for {feat}: {str(e)}")
                continue

        if valid_features > 0:
            avg_eodd /= valid_features
        else:
            avg_eodd = 0
        eodd_scores.append(avg_eodd)

        # Explainability: 1 - SHAP_Consistency_Gap
        # Sample 100 instances for SHAP computation
        if len(X_fold_val) > 100:
            sample_indices = np.random.choice(X_fold_val.index, 100, replace=False)
            X_sample = X_fold_val.loc[sample_indices]
        else:
            X_sample = X_fold_val

        shap_gap = calculate_shap_consistency_gap_tabnet(model, X_sample, sensitive_features)
        shap_gap_scores.append(shap_gap)

    # Aggregate scores across folds
    avg_mcc = np.mean(mcc_scores)
    avg_eodd = np.mean(eodd_scores)
    avg_shap_gap = np.mean(shap_gap_scores)

    # Return objectives: maximize MCC, maximize (1 - EODD), maximize (1 - SHAP_Gap)
    return avg_mcc, 1 - avg_eodd, 1 - avg_shap_gap

# Calculate Confidence Intervals (95%)
# ===================================================
def calculate_ci(data, confidence=0.95):
    a = 1.0 * np.array(data)
    n = len(a)
    m, se = np.mean(a), stats.sem(a)
    h = se * stats.t.ppf((1 + confidence) / 2., n-1)
    return m, m-h, m+h
